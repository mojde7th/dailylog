از صفحهٔ خالی تا DailyLog زنده

این کارگاه آموزش وب نیست. فرض این است که اچ‌تی‌ام‌ال و جاوااسکریپت را بلدی. هر نسخه یک تابع یا یک پروندهٔ کامل است که به قبلی اضافه می‌شود. جمله را دور شناسه خرد نمی‌کنیم. جعبه نمی‌کشیم. توضیح بعد از بلوک است، نه دو لایه روی هر خط.

اگر کارگاه و اپ زنده فرق داشتند، پرونده‌های زنده منبع‌اند. این صفحه آموزش است.

نسخهٔ 0 — صفحهٔ خالی که هنوز اپ نیست

هدف: یک پروندهٔ اچ‌تی‌ام‌ال که مرورگر بازش می‌کند. هنوز داده، تب، همگام، هیچ.

نسخهٔ 11 — باز کردن ایندکس‌دی‌بی

هدف: یک پایگاه، سه فروشگاه. نسخهٔ طرح را بی‌خودی بالا نبر.

function openDB() {
  if (_db) return Promise.resolve(_db);
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB_NAME, DB_VER);
    req.onupgradeneeded = ev => {
      const db = ev.target.result;
      if (!db.objectStoreNames.contains('sessions')) db.createObjectStore('sessions', { keyPath:'uid' });
      if (!db.objectStoreNames.contains('meta'))     db.createObjectStore('meta',     { keyPath:'uid' });
      if (!db.objectStoreNames.contains('config'))   db.createObjectStore('config',   { keyPath:'k' });
    };
    req.onsuccess = () => { _db = req.result; resolve(_db); };
    req.onerror   = () => reject(req.error);
  });
}

نشانی شیت در انبار محلی است نه اینجا. پاک‌کردن فروشگاه‌ها تنظیمات را لزوماً نمی‌کشد. اگر طرح را عوض کنی بدون مهاجرت، مرورگر فروشگاه خالی می‌سازد.

async function pushBatch(url, secret, type, rows, fields) {
  const payload = {
    secret, type, fields,
    rows: rows.map(r => fields.map(f => r[f] === undefined ? '' : r[f]))
  };
  const res = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body: JSON.stringify(payload)
  });
  const raw = await res.text();
  const outp = JSON.parse(raw);
  if (!outp.ok) throw new Error(outp.error || 'server');
  if (String(outp.version || '') !== SCRIPT_VERSION) {
    throw new Error('old script ' + (outp.version || ''));
  }
  const st = (type === 'meta') ? 'meta' : 'sessions';
  for (const r of rows) { r.synced = 1; await put(st, r); }
  return outp;
}
